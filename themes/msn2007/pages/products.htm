Developer SDK

Empowering third-party developers to extend the Coherent ecosystem with custom integrations.

import { SmartHome } from '@coherent/sdk';

// Initialize the local hub connection
const hub = await SmartHome.connect('local');

// Get the living room zone
const livingRoom = hub.getZone('living-room');

// Set scene to 'Movie Night'
await livingRoom.activateScene('movie-night', {
    transitionTime: 2000 // ms
});

console.log('Scene activated');
